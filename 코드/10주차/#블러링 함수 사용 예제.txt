#블러링 함수 사용 예제
import cv2
import numpy as np

BGR_img = cv2.imread("C:/Users/user/Desktop/images/sum_test.jpg",cv2.IMREAD_COLOR)

data = [ 1/9,1/9,1/9,
         1/9,1/9,1/9,
         1/9,1/9,1/9]
mask = np.array(data,np.float32).reshape(3,3)
blur3 = cv2.filter2D(BGR_img,-1,mask)
cv2.imshow("blur",blur3)
cv2.waitKey(0)

#샤프닝 예제
import cv2
import numpy as np

BGR_img = cv2.imread("C:/Users/user/Desktop/images/sum_test.jpg",cv2.IMREAD_COLOR)

data = [ 0,-1,0,
         -1,5,-1,
         0,-1,0]
data2 = [ -1,-1,-1,
         -1,9,-1,
         -1,-1,-1]
mask = np.array(data,np.float32).reshape(3,3)
mask2 = np.array(data2,np.float32)
sharpen1 = cv2.filter2D(BGR_img,-1,mask)
sharpen2 = cv2.filter2D(BGR_img,-1,mask2)
cv2.imshow("sharpen1",sharpen1)
cv2.imshow("sharpen2",sharpen2)
cv2.waitKey(0)

def My_Circle(img,p1,rad):
    for x in range(p1[0]-rad,p1[0]+rad+1,1):
        y1 = p1[1] + (rad**2 - (x-p1[1])**2) **0.5
        y2 = p1[1] - (rad**2 - (x-p1[1])**2) **0.5
        img[int(y1),x] = 255
        img[int(y2),x] = 255
